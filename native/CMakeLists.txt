cmake_minimum_required(VERSION 3.28)
project(FactorioAccessNative VERSION 0.1.0 LANGUAGES C CXX)

if(NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "The FactorioAccess native DLL builds only as a 64-bit MSVC DLL.")
endif()

set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
# Static CRT: the game directory gets one self-contained DLL and nothing to install beside it.
# Prism sets this itself from CMAKE_BUILD_TYPE, which is empty under the multi-config Visual
# Studio generator, so it is pinned in the cache before Prism is configured.
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>" CACHE STRING "" FORCE)

set(FA_FACTORIO_BIN "" CACHE PATH
    "Factorio bin/x64 directory; when set, every build copies winmm.dll there")

if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/vendor/prism/CMakeLists.txt")
  message(FATAL_ERROR "Submodules are missing; run: git submodule update --init native/vendor")
endif()

add_subdirectory(vendor/minhook EXCLUDE_FROM_ALL)

set(RAWPDB_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
add_subdirectory(vendor/raw_pdb EXCLUDE_FROM_ALL)

# Prism is built static so <prism.h> resolves to plain externs and the speech code lives inside
# our DLL.
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(PRISM_ENABLE_TESTS OFF CACHE BOOL "" FORCE)
set(PRISM_ENABLE_DEMOS OFF CACHE BOOL "" FORCE)
set(PRISM_ENABLE_SHIMS OFF CACHE BOOL "" FORCE)
set(PRISM_ENABLE_TOLK_SHIM OFF CACHE BOOL "" FORCE)
set(PRISM_ENABLE_GDEXTENSION OFF CACHE BOOL "" FORCE)
set(PRISM_ENABLE_LEGACY_BACKENDS OFF CACHE BOOL "" FORCE)
# UIA reports itself usable whenever COM can create CUIAutomation, listener or not, and ties with
# SAPI on priority, so a player without a screen reader could get silent UIA notifications instead
# of audible SAPI.
set(PRISM_ENABLE_UIA_BACKEND OFF CACHE STRING "" FORCE)
set(PRISM_ENABLE_POWER_MANAGEMENT OFF CACHE BOOL "" FORCE)
add_subdirectory(vendor/prism EXCLUDE_FROM_ALL)

# winmm.dll proxy: factorio.exe imports winmm and it is not a KnownDLL, so the copy in bin/x64 is
# the one the loader picks. Every export forwards to the system DLL by absolute path, so other
# modules that load winmm through us still get the real thing.
file(STRINGS src/proxy/winmm.exports _fa_exports REGEX "^[^;]")
set(_fa_forwarders "// Generated by CMakeLists.txt from src/proxy/winmm.exports.\n")
set(_fa_system_winmm [[\\\\.\\GLOBALROOT\\SystemRoot\\System32\\winmm]])
foreach(_fa_export IN LISTS _fa_exports)
  string(STRIP "${_fa_export}" _fa_export)
  if(_fa_export MATCHES "^@([0-9]+)$")
    set(_fa_ordinal "${CMAKE_MATCH_1}")
    string(APPEND _fa_forwarders
      "#pragma comment(linker, \"/export:fa_ordinal_${_fa_ordinal}=${_fa_system_winmm}.#${_fa_ordinal},@${_fa_ordinal},NONAME\")\n")
  else()
    string(APPEND _fa_forwarders
      "#pragma comment(linker, \"/export:${_fa_export}=${_fa_system_winmm}.${_fa_export}\")\n")
  endif()
endforeach()
file(CONFIGURE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/winmm_forwarders.cpp"
     CONTENT "${_fa_forwarders}" @ONLY)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS src/proxy/winmm.exports)

# Everything that reads the game without running in it, shared with the tools.
add_library(fa_core STATIC
  src/log.cpp
  src/symbols.cpp
  src/game.cpp
)
target_include_directories(fa_core PUBLIC src)
target_compile_definitions(fa_core PUBLIC WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE)
target_compile_options(fa_core PUBLIC /W4 /permissive- /utf-8 /EHsc)
target_link_libraries(fa_core PRIVATE raw_pdb)

# The Graph A11y Kernel (spec: WH40KRTAccess/docs/graph-a11y-spec.md), STL only. The conformance
# suite compiles it with no other include path, which is what keeps game code out of it.
set(FA_GRAPH_SOURCES
  src/graph/GraphBuilder.cpp
  src/graph/KeyGraph.cpp
  src/graph/GraphAnnouncer.cpp
)
add_library(fa_graph STATIC ${FA_GRAPH_SOURCES})
target_include_directories(fa_graph PUBLIC src)
target_compile_options(fa_graph PRIVATE /W4 /permissive- /utf-8 /EHsc)
target_link_libraries(fa_native PRIVATE fa_graph)
